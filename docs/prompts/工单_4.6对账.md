## 五、附带脚本（在本机执行）

### `count_version.py`

```python
# 放在 G:\HSR 根目录执行：python count_version.py
import pathlib, collections
root = pathlib.Path("zh_cn")
stat = collections.Counter()
for p in root.rglob("*.md"):
    ver = "无"
    for line in p.read_text(encoding="utf-8", errors="ignore").splitlines()[:8]:
        if line.startswith("> 数据版本："):
            ver = line.replace("> 数据版本：", "").strip()
            break
    parts = p.relative_to(root).parts
    key = "/".join(parts[:2]) if len(parts) > 1 else parts[0]
    stat[(key, ver)] += 1
for (key, ver), n in sorted(stat.items()):
    print(f"{key}\t{ver}\t{n}")
```

> 期望：`zh_cn` 合计 4.6 = **2454**、4.5 = **3847**。若与此不符，说明扫描范围或跳过目录不同，先报告差异。

### `fix_obtain_list.py`（**本工单只做 dry-run，不实际修复**）

```python
# 用法：python fix_obtain_list.py --dry-run
import ast, pathlib, re, sys
dry = "--dry-run" in sys.argv
pat = re.compile(r"^- (\[.*\])$")
fixed = 0
for p in pathlib.Path("zh_cn").rglob("*.md"):
    txt = p.read_text(encoding="utf-8")
    if not pat.search(txt):
        continue
    out, changed = [], False
    for line in txt.splitlines():
        m = pat.match(line)
        if m:
            try:
                items = ast.literal_eval(m.group(1))
            except Exception:
                out.append(line); continue
            out.extend(f"- {str(i).strip()}" for i in items)
            changed = True
        else:
            out.append(line)
    if changed:
        fixed += 1
        if not dry:
            p.write_text("\n".join(out) + "\n", encoding="utf-8")
        print(("DRY " if dry else "FIX ") + str(p))
print(f"\n{'将修复' if dry else '已修复'} {fixed} 个文件")
```
